if(NINTENDO_SWITCH)
    add_library(savetinker_platform STATIC switch/switch_platform.cpp)
    target_include_directories(savetinker_platform PUBLIC switch)
    if(NINTENDO_SWITCH2)
        target_compile_definitions(savetinker_platform PRIVATE SAVETINKER_SWITCH2=1)
    endif()
elseif(NINTENDO_3DS)
    add_library(savetinker_platform STATIC 3ds/ctr_platform.cpp 3ds/ds_card.cpp)
    target_include_directories(savetinker_platform PUBLIC 3ds)
    target_link_libraries(savetinker_platform PUBLIC ctru)
else()
    add_library(savetinker_platform STATIC pc/pc_platform.cpp)
    target_include_directories(savetinker_platform PUBLIC pc)
endif()
target_link_libraries(savetinker_platform PUBLIC savetinker_core)
target_compile_options(savetinker_platform PRIVATE -Wall -Wextra)
